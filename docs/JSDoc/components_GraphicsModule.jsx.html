/**
 * @typedef {Object} ChartSummaryStats
 * Resumen estadístico del conjunto de categorías visibles en el gráfico.
 * @property {number} total - Suma total de los valores agregados.
 * @property {number} avg - Media aritmética calculada sobre las categorías graficadas.
 * @property {string} topCategory - Nombre de la categoría con valor más alto.
 * @property {number} topValue - Valor numérico de la categoría líder.

 * @property {number} count - Número total de elementos mostrados.

/**
 * Componente que provee un estudio interactivo de gráficos con Recharts.
 * Permite alternar tipos de visualización, agrupar por cualquier columna, calcular agregaciones
 * estadísticas, personalizar temas cromáticos y descargar el gráfico resultante en formato PNG.
 *
 * @component
 * @returns {JSX.Element|null} Panel interactivo de gráficos o `null` si no hay datos disponibles.
 */
export default function GraphicsModule() {
  const { processedData, filteredData, sortConfig, columns, fileName } = useData();
  const { isDark } = useTheme();
  const [isOpen, setIsOpen] = useState(true);
  const chartContainerRef = useRef(null);
  const [isExporting, setIsExporting] = useState(false);
  const [copiedSuccess, setCopiedSuccess] = useState(false);

  const currentData = useMemo(() => {
    return filteredData !== null && filteredData !== undefined
      ? filteredData
      : processedData || [];
  }, [filteredData, processedData]);

  // Columnas numéricas identificadas
  const numericColumns = useMemo(() => {
    if (!currentData || currentData.length === 0) return [];
    return columns.filter((col) => {
      const numericCount = currentData.filter((row) => {
        const val = row[col];
        return (
          val !== null &&
          val !== undefined &&
          val !== "" &&
          !isNaN(Number(val)) &&
          typeof val !== "boolean"
        );
      }).length;
      return numericCount > currentData.length * 0.4;
    });
  }, [currentData, columns]);

  // Estados de configuración de visualización
  const [chartType, setChartType] = useState("bar"); // 'bar', 'bar-horizontal', 'line', 'area', 'pie', 'donut', 'composed'
  const [userCategoryCol, setUserCategoryCol] = useState("");
  const categoryCol =
    userCategoryCol && columns.includes(userCategoryCol)
      ? userCategoryCol
      : columns[0] || "";

  const [userMetricCol, setUserMetricCol] = useState("");
  const metricCol =
    userMetricCol && numericColumns.includes(userMetricCol)
      ? userMetricCol
      : numericColumns[0] || "";

  const [userSecondaryMetricCol, setUserSecondaryMetricCol] = useState("");
  const secondaryMetricCol =
    userSecondaryMetricCol &&
      numericColumns.includes(userSecondaryMetricCol) &&
      userSecondaryMetricCol !== metricCol
      ? userSecondaryMetricCol
      : "";

  const [rawAggregation, setRawAggregation] = useState("sum"); // 'sum', 'avg', 'count', 'max', 'min'
  const aggregation = numericColumns.length === 0 ? "count" : rawAggregation;

  const [limit, setLimit] = useState(10); // 5, 10, 15, 20, 0 (todos)
  const [sortBy, setSortBy] = useState("desc"); // 'desc', 'asc', 'none'
  const [selectedPalette, setSelectedPalette] = useState("emerald");
  const [showGrid, setShowGrid] = useState(true);
  const [showLegend, setShowLegend] = useState(true);

  /**
   * Cálculo memorizado y agregación de métricas para la renderización del gráfico en Recharts.
   * Aplica conteos, sumas, promedios, valores mínimos o máximos, y sincroniza el orden
   * si existe un criterio activo proveniente de DataExplorer.
   */
  const aggregatedData = useMemo(() => {
    if (!currentData || currentData.length === 0 || !categoryCol) return [];

    const grouped = new Map();

    currentData.forEach((row) => {
      const rawCategory = row[categoryCol];
      const categoryKey =
        rawCategory !== null && rawCategory !== undefined && rawCategory !== ""
          ? String(rawCategory).trim()
          : "(Vacío)";

      const val1 = metricCol ? Number(row[metricCol]) : 0;
      const val2 = secondaryMetricCol ? Number(row[secondaryMetricCol]) : 0;

      if (!grouped.has(categoryKey)) {
        grouped.set(categoryKey, {
          name: categoryKey,
          items: 0,
          sum1: 0,
          values1: [],
          sum2: 0,
          values2: [],
        });
      }

      const entry = grouped.get(categoryKey);
      entry.items += 1;

      if (!isNaN(val1)) {
        entry.sum1 += val1;
        entry.values1.push(val1);
      }
      if (!isNaN(val2) && secondaryMetricCol) {
        entry.sum2 += val2;
        entry.values2.push(val2);
      }
    });

    let result = Array.from(grouped.values()).map((entry) => {
      let finalVal1 = 0;
      let finalVal2 = 0;

      if (aggregation === "count") {
        finalVal1 = entry.items;
        finalVal2 = entry.items;
      } else if (aggregation === "sum") {
        finalVal1 = entry.sum1;
        finalVal2 = entry.sum2;
      } else if (aggregation === "avg") {
        finalVal1 = entry.values1.length ? entry.sum1 / entry.values1.length : 0;
        finalVal2 = entry.values2.length ? entry.sum2 / entry.values2.length : 0;
      } else if (aggregation === "max") {
        finalVal1 = entry.values1.length ? Math.max(...entry.values1) : 0;
        finalVal2 = entry.values2.length ? Math.max(...entry.values2) : 0;
      } else if (aggregation === "min") {
        finalVal1 = entry.values1.length ? Math.min(...entry.values1) : 0;
        finalVal2 = entry.values2.length ? Math.min(...entry.values2) : 0;
      }

      const item = {
        name: entry.name,
        displayName:
          entry.name.length > 20 ? entry.name.slice(0, 18) + "..." : entry.name,
        value: Number(finalVal1.toFixed(2)),
        itemsCount: entry.items,
      };

      if (secondaryMetricCol) {
        item.value2 = Number(finalVal2.toFixed(2));
      }

      return item;
    });

    // Ordenamiento: si hay orden activo en DataExplorer, se sincroniza el gráfico
    if (sortConfig?.key) {
      if (sortConfig.key === categoryCol) {
        result.sort((a, b) => {
          const strA = String(a.name).toLowerCase();
          const strB = String(b.name).toLowerCase();
          return sortConfig.direction === "asc"
            ? strA.localeCompare(strB)
            : strB.localeCompare(strA);
        });
      } else if (sortConfig.key === metricCol) {
        result.sort((a, b) =>
          sortConfig.direction === "asc" ? a.value - b.value : b.value - a.value
        );
      } else {
        if (sortBy === "desc") {
          result.sort((a, b) => b.value - a.value);
        } else if (sortBy === "asc") {
          result.sort((a, b) => a.value - b.value);
        }
      }
    } else if (sortBy === "desc") {
      result.sort((a, b) => b.value - a.value);
    } else if (sortBy === "asc") {
      result.sort((a, b) => a.value - b.value);
    }

    // Límite Top N
    if (limit > 0 && result.length > limit) {
      result = result.slice(0, limit);
    }

    return result;
  }, [
    currentData,
    categoryCol,
    metricCol,
    secondaryMetricCol,
    aggregation,
    sortBy,
    sortConfig,
    limit,
  ]);

  /**
   * Resumen estadístico dinámico de las categorías y valores presentes en el gráfico activo.
   *
   * @type {ChartSummaryStats}
   */
  const chartSummary = useMemo(() => {
    if (aggregatedData.length === 0) {
      return { total: 0, avg: 0, topCategory: "N/A", topValue: 0, count: 0 };
    }

    const total = aggregatedData.reduce((acc, curr) => acc + curr.value, 0);
    const avg = total / aggregatedData.length;
    const topItem = [...aggregatedData].sort((a, b) => b.value - a.value)[0];

    return {
      total: Number(total.toFixed(2)),
      avg: Number(avg.toFixed(2)),
      topCategory: topItem ? topItem.name : "N/A",
      topValue: topItem ? topItem.value : 0,
      topPercent:
        total > 0 && topItem
          ? ((topItem.value / total) * 100).toFixed(1)
          : "0.0",
      count: aggregatedData.length,
    };
  }, [aggregatedData]);

  // Colores activos
  const activePalette = COLOR_PALETTES[selectedPalette] || COLOR_PALETTES.emerald;

  /**
   * Exporta el elemento SVG del gráfico visible a una imagen PNG en alta definición (2x).
   * Clona el nodo vectorial, computa e inyecta los estilos CSS en línea, renderiza el gráfico
   * en un elemento `<canvas>` y desencadena la descarga del archivo en el navegador.
   *
   * @async
   * @returns {Promise<void>}
   */
  const handleDownloadChart = async () => {
    const container = chartContainerRef.current;
    if (!container || isExporting) return;

    // No usar querySelector("svg") a secas: puede devolver el SVG de un icono
    // (por ejemplo, 14 × 14) en lugar del SVG que dibuja el gráfico.
    const svgCandidates = Array.from(container.querySelectorAll("svg"));
    const getSvgDimensions = (svg) => {
      const viewBox = svg.viewBox?.baseVal;
      const rect = svg.getBoundingClientRect();
      const widthAttr = Number.parseFloat(svg.getAttribute("width"));
      const heightAttr = Number.parseFloat(svg.getAttribute("height"));
      return {
        width: Math.round((Number.isFinite(widthAttr) && widthAttr > 0 ? widthAttr : 0) || viewBox?.width || rect.width || 0),
        height: Math.round((Number.isFinite(heightAttr) && heightAttr > 0 ? heightAttr : 0) || viewBox?.height || rect.height || 0),
      };
    };
    const validCandidates = svgCandidates
      .map((svg) => ({ svg, ...getSvgDimensions(svg) }))
      .filter(({ width, height }) => width >= 100 && height >= 100)
      .sort((a, b) => {
        const aSurface = a.svg.classList.contains("recharts-surface") ? 1 : 0;
        const bSurface = b.svg.classList.contains("recharts-surface") ? 1 : 0;
        if (aSurface !== bSurface) return bSurface - aSurface;
        return (b.width * b.height) - (a.width * a.height);
      });
    const selected = validCandidates[0];
    const svgElement = selected?.svg;

    if (!svgElement) {
      const sizes = svgCandidates.map((svg) => {
        const { width, height } = getSvgDimensions(svg);
        return `${width} × ${height} (${svg.getAttribute("class") || "sin clase"})`;
      });
      console.error("No se encontró un SVG de gráfico con dimensiones válidas.", sizes);
      window.alert("No se encontró el gráfico para exportar. Comprueba que el gráfico esté visible y vuelve a intentarlo.");
      return;
    }

    setIsExporting(true);
    let svgUrl;

    try {
      // Las dimensiones corresponden al SVG del gráfico, nunca a un icono pequeño.
      const { width, height } = selected;

      const clone = svgElement.cloneNode(true);
      clone.setAttribute("xmlns", "http://www.w3.org/2000/svg");
      clone.setAttribute("width", String(width));
      clone.setAttribute("height", String(height));
      if (!clone.hasAttribute("viewBox")) {
        clone.setAttribute("viewBox", `0 0 ${width} ${height}`);
      }

      // Copiar estilos calculados al SVG exportado para que no dependa del CSS de la página.
      const originalNodes = [svgElement, ...svgElement.querySelectorAll("*")];
      const clonedNodes = [clone, ...clone.querySelectorAll("*")];
      originalNodes.forEach((node, index) => {
        const target = clonedNodes[index];
        if (!target) return;
        const computed = window.getComputedStyle(node);
        const properties = [
          "color", "font-family", "font-size", "font-weight", "font-style",
          "fill", "fill-opacity", "stroke", "stroke-width", "stroke-dasharray",
          "stroke-linecap", "stroke-linejoin", "opacity", "text-anchor",
          "dominant-baseline", "visibility", "display"
        ];
        properties.forEach((property) => {
          const value = computed.getPropertyValue(property);
          if (value) target.style.setProperty(property, value);
        });
      });

      const svgBlob = new Blob(
        [new XMLSerializer().serializeToString(clone)],
        { type: "image/svg+xml;charset=utf-8" }
      );
      svgUrl = URL.createObjectURL(svgBlob);

      const image = new Image();
      image.decoding = "async";
      await new Promise((resolve, reject) => {
        image.onload = resolve;
        image.onerror = () => reject(new Error("El navegador no pudo procesar el SVG del gráfico."));
        image.src = svgUrl;
      });

      const scale = 2;
      const canvas = document.createElement("canvas");
      canvas.width = width * scale;
      canvas.height = height * scale;
      const context = canvas.getContext("2d");
      if (!context) throw new Error("No se pudo crear el contexto del canvas.");

      context.fillStyle = "#ffffff";
      context.fillRect(0, 0, canvas.width, canvas.height);
      context.scale(scale, scale);
      context.drawImage(image, 0, 0, width, height);

      const pngBlob = await new Promise((resolve, reject) => {
        canvas.toBlob((blob) => {
          if (!blob || blob.size < 1000) {
            reject(new Error(`El PNG generado está vacío o es demasiado pequeño (${blob?.size ?? 0} bytes).`));
          } else {
            resolve(blob);
          }
        }, "image/png");
      });

      const pngUrl = URL.createObjectURL(pngBlob);
      const link = document.createElement("a");
      const baseName = fileName ? fileName.replace(/\.[^/.]+$/, "") : "analisis";
      link.href = pngUrl;
      link.download = `grafico-${baseName}-${chartType}.png`;
      document.body.appendChild(link);
      link.click();
      link.remove();
      window.setTimeout(() => URL.revokeObjectURL(pngUrl), 1500);

      setCopiedSuccess(true);
      window.setTimeout(() => setCopiedSuccess(false), 3000);
    } catch (error) {
      console.error("Error al exportar gráfico como PNG:", error);
      window.alert(`No se pudo exportar el gráfico. ${error.message || "Revisá la consola del navegador para más detalles."}`);
    } finally {
      if (svgUrl) URL.revokeObjectURL(svgUrl);
      setIsExporting(false);
    }
  };

  if (!processedData || processedData.length === 0) return null;

  return (
    <div className="w-full bg-white dark:bg-zinc-900 border border-slate-200 dark:border-zinc-800 rounded-xl shadow-sm overflow-hidden mb-6">
      {/* Header Desplegable */}
      <div
        onClick={() => setIsOpen(!isOpen)}
        className="p-4 bg-slate-50 dark:bg-zinc-900/90 border-b border-slate-200 dark:border-zinc-800 flex items-center justify-between cursor-pointer hover:bg-slate-100/80 dark:hover:bg-zinc-800/80 transition-colors"
      >
        <div className="flex items-center gap-3">
          <div className="p-2 bg-emerald-100 dark:bg-emerald-950/70 text-emerald-700 dark:text-emerald-400 rounded-lg">
            <BarChart3 className="w-5 h-5" />
          </div>
          <div>
            <div className="flex items-center gap-2">
              <h3 className="font-semibold text-slate-800 dark:text-zinc-100 text-sm">
                Módulo de Gráficos Interactivos
              </h3>
              {filteredData !== null && filteredData !== undefined && (
                <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-medium bg-emerald-100 dark:bg-emerald-950/60 text-emerald-800 dark:text-emerald-300">
                  <Filter className="w-3 h-3" />
                  Sincronizado con filtro ({currentData.length} filas)
                </span>
              )}
            </div>
            <p className="text-xs text-slate-500 dark:text-zinc-400">
              Visualización dinámica, métricas comparativas y exportación de
              diagramas
            </p>
          </div>
        </div>

        <div className="flex items-center gap-2">
          {isOpen ? (
            <ChevronUp className="w-4 h-4 text-slate-400 dark:text-zinc-500" />
          ) : (
            <ChevronDown className="w-4 h-4 text-slate-400 dark:text-zinc-500" />
          )}
        </div>
      </div>

      {isOpen && (
        <div className="p-6 space-y-6">
          {/* Panel de Controles / Toolbar */}
          <div className="p-4 bg-slate-50 dark:bg-zinc-950/70 border border-slate-200 dark:border-zinc-800 rounded-xl space-y-4">
            <div className="flex flex-col lg:flex-row lg:items-center justify-between gap-3 border-b border-slate-200/70 dark:border-zinc-800 pb-3">
              {/* Selector de Tipo de Gráfico con Botones */}
              <div className="flex items-center gap-1.5 flex-wrap">
                <span className="text-xs font-semibold text-slate-500 dark:text-zinc-400 mr-1 flex items-center gap-1">
                  <Layers className="w-3.5 h-3.5 text-emerald-600 dark:text-emerald-400" />
                  Tipo:
                </span>
                {[
                  { id: "bar", label: "Barras", icon: BarChart3 },
                  { id: "bar-horizontal", label: "Horizontal", icon: BarChart3 },
                  { id: "line", label: "Líneas", icon: LineChartIcon },
                  { id: "area", label: "Áreas", icon: Sparkles },
                  { id: "pie", label: "Pastel", icon: PieChartIcon },
                  { id: "donut", label: "Dona", icon: PieChartIcon },
                  { id: "composed", label: "Compuesto", icon: TrendingUp },
                ].map((type) => {
                  const Icon = type.icon;
                  const isActive = chartType === type.id;
                  return (
                    <button
                      key={type.id}
                      type="button"
                      onClick={() => setChartType(type.id)}
                      className={`inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-medium transition-all ${isActive
                        ? "bg-emerald-600 text-white shadow-sm shadow-emerald-200 dark:shadow-none font-semibold"
                        : "bg-white dark:bg-zinc-800 text-slate-600 dark:text-zinc-300 border border-slate-200 dark:border-zinc-700 hover:bg-slate-100 dark:hover:bg-zinc-700 hover:text-slate-800 dark:hover:text-white"
                        }`}
                    >
                      <Icon className="w-3.5 h-3.5" />
                      {type.label}
                    </button>
                  );
                })}
              </div>

              {/* Botón de Exportación rápida */}
              <div className="flex items-center gap-2">
                <button
                  type="button"
                  onClick={handleDownloadChart}
                  disabled={isExporting || aggregatedData.length === 0}
                  className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg text-xs font-medium bg-white dark:bg-zinc-800 border border-slate-200 dark:border-zinc-700 text-slate-700 dark:text-zinc-200 hover:bg-slate-100 dark:hover:bg-zinc-700 hover:border-slate-300 dark:hover:border-zinc-600 disabled:opacity-50 transition-colors"
                  title="Descargar este gráfico en formato de imagen PNG de alta calidad"
                >
                  {copiedSuccess ? (
                    <>
                      <Check className="w-3.5 h-3.5 text-emerald-600 dark:text-emerald-400" />
                      <span className="text-emerald-600 dark:text-emerald-400 font-semibold">
                        ¡Descargado!
                      </span>
                    </>
                  ) : (
                    <>
                      <Download className="w-3.5 h-3.5 text-slate-500 dark:text-zinc-400" />
                      <span>Exportar PNG</span>
                    </>
                  )}
                </button>
              </div>
            </div>

            {/* Selectores de dimensiones y métricas */}
            <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 lg:grid-cols-6 gap-3 text-xs">
              {/* Eje X / Categoría */}
              <div>
                <label htmlFor="chart-category-col" className="block text-slate-600 dark:text-zinc-300 font-medium mb-1 truncate">
                  Eje X (Categoría):
                </label>
                <select
                  id="chart-category-col"
                  value={categoryCol}
                  onChange={(e) => setUserCategoryCol(e.target.value)}
                  className="w-full px-2.5 py-1.5 bg-white dark:bg-zinc-800 border border-slate-300 dark:border-zinc-700 text-slate-800 dark:text-zinc-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500"
                >
                  {columns.map((col) => (
                    <option key={col} value={col}>
                      {col}
                    </option>
                  ))}
                </select>
              </div>

              {/* Métrica Principal */}
              <div>
                <label htmlFor="chart-metric-col" className="block text-slate-600 dark:text-zinc-300 font-medium mb-1 truncate">
                  Métrica Principal (Y):
                </label>
                <select
                  id="chart-metric-col"
                  value={metricCol}
                  onChange={(e) => setUserMetricCol(e.target.value)}
                  disabled={aggregation === "count"}
                  className="w-full px-2.5 py-1.5 bg-white dark:bg-zinc-800 border border-slate-300 dark:border-zinc-700 text-slate-800 dark:text-zinc-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500 disabled:bg-slate-100 dark:disabled:bg-zinc-850 disabled:text-slate-400 dark:disabled:text-zinc-600"
                >
                  {numericColumns.length > 0 ? (
                    numericColumns.map((col) => (
                      <option key={col} value={col}>
                        {col}
                      </option>
                    ))
                  ) : (
                    <option value="">(Sin columnas numéricas)</option>
                  )}
                </select>
              </div>

              {/* Función de Agregación */}
              <div>
                <label htmlFor="chart-aggregation" className="block text-slate-600 dark:text-zinc-300 font-medium mb-1 truncate">
                  Agregación:
                </label>
                <select
                  id="chart-aggregation"
                  value={aggregation}
                  onChange={(e) => setRawAggregation(e.target.value)}
                  className="w-full px-2.5 py-1.5 bg-white dark:bg-zinc-800 border border-slate-300 dark:border-zinc-700 text-slate-800 dark:text-zinc-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500"
                >
                  <option value="sum">Suma total</option>
                  <option value="avg">Promedio</option>
                  <option value="count">Conteo de filas</option>
                  <option value="max">Valor Máximo</option>
                  <option value="min">Valor Mínimo</option>
                </select>
              </div>

              {/* Segunda Métrica Comparativa (Opcional) */}
              <div>
                <label htmlFor="chart-secondary-metric" className="block text-slate-600 dark:text-zinc-300 font-medium mb-1 truncate">
                  Métrica 2 (Opcional):
                </label>
                <select
                  id="chart-secondary-metric"
                  value={secondaryMetricCol}
                  onChange={(e) => setUserSecondaryMetricCol(e.target.value)}
                  disabled={
                    aggregation === "count" ||
                    chartType === "pie" ||
                    chartType === "donut"
                  }
                  className="w-full px-2.5 py-1.5 bg-white dark:bg-zinc-800 border border-slate-300 dark:border-zinc-700 text-slate-800 dark:text-zinc-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500 disabled:bg-slate-100 dark:disabled:bg-zinc-850 disabled:text-slate-400 dark:disabled:text-zinc-600"
                >
                  <option value="">(Ninguna)</option>
                  {numericColumns
                    .filter((col) => col !== metricCol)
                    .map((col) => (
                      <option key={col} value={col}>
                        {col}
                      </option>
                    ))}
                </select>
              </div>

              {/* Límite Top N */}
              <div>
                <label htmlFor="chart-top-n" className="block text-slate-600 dark:text-zinc-300 font-medium mb-1 truncate">
                  Mostrar (Top N):
                </label>
                <select
                  id="chart-top-n"
                  value={limit}
                  onChange={(e) => setLimit(Number(e.target.value))}
                  className="w-full px-2.5 py-1.5 bg-white dark:bg-zinc-800 border border-slate-300 dark:border-zinc-700 text-slate-800 dark:text-zinc-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500"
                >
                  <option value={5}>Top 5 elementos</option>
                  <option value={10}>Top 10 elementos</option>
                  <option value={15}>Top 15 elementos</option>
                  <option value={20}>Top 20 elementos</option>
                  <option value={0}>Todos los elementos</option>
                </select>
              </div>

              {/* Ordenamiento */}
              <div>
                <label htmlFor="chart-sort-order" className="block text-slate-600 dark:text-zinc-300 font-medium mb-1 truncate">
                  Orden de Datos:
                </label>
                <select
                  id="chart-sort-order"
                  value={sortBy}
                  onChange={(e) => setSortBy(e.target.value)}
                  className="w-full px-2.5 py-1.5 bg-white dark:bg-zinc-800 border border-slate-300 dark:border-zinc-700 text-slate-800 dark:text-zinc-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-emerald-500"
                >
                  <option value="desc">Mayor a menor</option>
                  <option value="asc">Menor a mayor</option>
                  <option value="none">Orden original</option>
                </select>
              </div>
            </div>

            {/* Ajustes visuales: Paleta y Toggles */}
            <div className="flex flex-wrap items-center justify-between gap-4 pt-2 border-t border-slate-200/70 dark:border-zinc-800 text-xs">
              <div className="flex items-center gap-3">
                <span className="text-slate-500 dark:text-zinc-400 font-medium flex items-center gap-1">
                  <Palette className="w-3.5 h-3.5 text-slate-400 dark:text-zinc-500" />
                  Paleta de color:
                </span>
                <div className="flex items-center gap-1.5">
                  {Object.entries(COLOR_PALETTES).map(([key, pal]) => (
                    <button
                      key={key}
                      type="button"
                      onClick={() => setSelectedPalette(key)}
                      title={pal.name}
                      className={`h-6 px-2 rounded-md flex items-center gap-1.5 transition-all text-[11px] ${selectedPalette === key
                        ? "bg-white dark:bg-zinc-800 border-2 border-slate-800 dark:border-zinc-200 text-slate-900 dark:text-zinc-100 font-semibold shadow-xs"
                        : "bg-white/80 dark:bg-zinc-900 border border-slate-200 dark:border-zinc-700 hover:border-slate-300 dark:hover:border-zinc-600 text-slate-600 dark:text-zinc-400"
                        }`}
                    >
                      <span
                        className="w-2.5 h-2.5 rounded-full inline-block shrink-0"
                        style={{ backgroundColor: pal.primary }}
                      />
                      <span className="hidden sm:inline">{pal.name}</span>
                    </button>
                  ))}
                </div>
              </div>

              <div className="flex items-center gap-4 text-slate-600 dark:text-zinc-300">
                <label className="flex items-center gap-1.5 cursor-pointer">
                  <input
                    type="checkbox"
                    checked={showGrid}
                    onChange={(e) => setShowGrid(e.target.checked)}
                    className="rounded border-slate-300 dark:border-zinc-700 text-emerald-600 focus:ring-emerald-500"
                  />
                  <span>Cuadrícula</span>
                </label>

                <label className="flex items-center gap-1.5 cursor-pointer">
                  <input
                    type="checkbox"
                    checked={showLegend}
                    onChange={(e) => setShowLegend(e.target.checked)}
                    className="rounded border-slate-300 dark:border-zinc-700 text-emerald-600 focus:ring-emerald-500"
                  />
                  <span>Leyenda</span>
                </label>
              </div>
            </div>
          </div>

          {/* Tarjetas KPI de Resumen del Gráfico */}
          <div className="grid grid-cols-2 md:grid-cols-4 gap-3 text-xs">
            <div className="p-3.5 bg-slate-50/80 dark:bg-zinc-950/60 border border-slate-200/70 dark:border-zinc-800 rounded-xl">
              <span className="text-slate-500 dark:text-zinc-400 font-medium block mb-1">
                Total Agregado ({aggregation === "count" ? "Filas" : metricCol || "Métrica"})
              </span>
              <span className="text-lg font-bold text-slate-800 dark:text-zinc-100">
                {chartSummary.total.toLocaleString(undefined, {
                  maximumFractionDigits: 2,
                })}
              </span>
            </div>

            <div className="p-3.5 bg-slate-50/80 dark:bg-zinc-950/60 border border-slate-200/70 dark:border-zinc-800 rounded-xl">
              <span className="text-slate-500 dark:text-zinc-400 font-medium block mb-1">
                Promedio por Categoría
              </span>
              <span className="text-lg font-bold text-slate-800 dark:text-zinc-100">
                {chartSummary.avg.toLocaleString(undefined, {
                  maximumFractionDigits: 2,
                })}
              </span>
            </div>

            <div className="p-3.5 bg-emerald-50/70 dark:bg-emerald-950/30 border border-emerald-100 dark:border-emerald-900/50 rounded-xl">
              <span className="text-emerald-700 dark:text-emerald-400 font-medium block mb-1 truncate">
                Líder: {chartSummary.topCategory}
              </span>
              <div className="flex items-baseline gap-2">
                <span className="text-lg font-bold text-emerald-900 dark:text-emerald-200">
                  {chartSummary.topValue.toLocaleString(undefined, {
                    maximumFractionDigits: 2,
                  })}
                </span>
                <span className="text-[11px] font-semibold text-emerald-600 dark:text-emerald-400">
                  ({chartSummary.topPercent}%)
                </span>
              </div>
            </div>

            <div className="p-3.5 bg-slate-50/80 dark:bg-zinc-950/60 border border-slate-200/70 dark:border-zinc-800 rounded-xl">
              <span className="text-slate-500 dark:text-zinc-400 font-medium block mb-1">
                Categorías Graficadas
              </span>
              <span className="text-lg font-bold text-slate-800 dark:text-zinc-100">
                {chartSummary.count}
              </span>
            </div>
          </div>

          {/* Área de Visualización Gráfica Recharts */}
          <div
            ref={chartContainerRef}
            className="w-full bg-slate-50/40 dark:bg-zinc-950/60 border border-slate-200/80 dark:border-zinc-800 rounded-xl p-4 sm:p-6"
          >

            {aggregatedData.length > 0 ? (
              <div className="w-full h-[400px]">
                <ResponsiveContainer width="100%" height="100%">
                  {/* Renderizado condicional según tipo de gráfico */}
                  {chartType === "bar" && (
                    <BarChart
                      data={aggregatedData}
                      margin={{ top: 20, right: 30, left: 20, bottom: 60 }}
                    >
                      {showGrid && (
                        <CartesianGrid
                          strokeDasharray="3 3"
                          stroke={isDark ? "#27272a" : "#e2e8f0"}
                          vertical={false}
                        />
                      )}
                      <XAxis
                        dataKey="displayName"
                        stroke={isDark ? "#a1a1aa" : "#64748b"}
                        fontSize={11}
                        interval={0}
                        angle={-25}
                        textAnchor="end"
                        tickMargin={10}
                      />
                      <YAxis
                        stroke={isDark ? "#a1a1aa" : "#64748b"}
                        fontSize={11}
                        tickFormatter={(v) =>
                          v >= 1000 ? `${(v / 1000).toFixed(0)}k` : v
                        }
                      />
                      <Tooltip
                        contentStyle={{
                          backgroundColor: isDark ? "#27272a" : "#fff",
                          color: isDark ? "#fff" : "#000",
                          borderRadius: "8px",
                          border: isDark ? "1px solid #27272a" : "1px solid #e2e8f0",
                          fontSize: "12px",
                        }}
                        cursor={{ fill: isDark ? "#27272a" : "#e2e8f0" }} />
                      {showLegend && <Legend verticalAlign="top" height={36} />}
                      <Bar
                        dataKey="value"
                        name={metricCol || "Total"}
                        fill={activePalette.primary}
                        radius={[6, 6, 0, 0]}
                        animationDuration={600}
                      >
                        {selectedPalette === "vibrant" &&
                          aggregatedData.map((entry, index) => (
                            <Cell
                              key={`cell-${index}`}
                              fill={
                                activePalette.colors[
                                index % activePalette.colors.length
                                ]
                              }
                            />
                          ))}
                      </Bar>
                      {secondaryMetricCol && (
                        <Bar
                          dataKey="value2"
                          name={secondaryMetricCol}
                          fill={activePalette.secondary}
                          radius={[6, 6, 0, 0]}
                          animationDuration={600}
                        />
                      )}
                    </BarChart>
                  )}

                  {chartType === "bar-horizontal" && (
                    <BarChart
                      layout="vertical"
                      data={aggregatedData}
                      margin={{ top: 20, right: 30, left: 60, bottom: 20 }}
                    >
                      {showGrid && (
                        <CartesianGrid
                          strokeDasharray="3 3"
                          stroke={isDark ? "#27272a" : "#e2e8f0"}
                          horizontal={false}
                        />
                      )}
                      <XAxis
                        type="number"
                        stroke={isDark ? "#a1a1aa" : "#64748b"}
                        fontSize={11}
                        tickFormatter={(v) =>
                          v >= 1000 ? `${(v / 1000).toFixed(0)}k` : v
                        }
                      />
                      <YAxis
                        dataKey="displayName"
                        type="category"
                        stroke={isDark ? "#a1a1aa" : "#64748b"}
                        fontSize={11}
                        width={90}
                      />
                      <Tooltip
                        contentStyle={{
                          backgroundColor: isDark ? "#27272a" : "#fff",
                          color: isDark ? "#fff" : "#000",
                          borderRadius: "8px",
                          border: isDark ? "1px solid #27272a" : "1px solid #e2e8f0",
                          fontSize: "12px",
                        }}
                        cursor={{ fill: isDark ? "#27272a" : "#e2e8f0" }} />
                      {showLegend && <Legend verticalAlign="top" height={36} />}
                      <Bar
                        dataKey="value"
                        name={metricCol || "Total"}
                        fill={activePalette.primary}
                        radius={[0, 6, 6, 0]}
                        animationDuration={600}
                      >
                        {selectedPalette === "vibrant" &&
                          aggregatedData.map((entry, index) => (
                            <Cell
                              key={`cell-${index}`}
                              fill={
                                activePalette.colors[
                                index % activePalette.colors.length
                                ]
                              }
                            />
                          ))}
                      </Bar>
                      {secondaryMetricCol && (
                        <Bar
                          dataKey="value2"
                          name={secondaryMetricCol}
                          fill={activePalette.secondary}
                          radius={[0, 6, 6, 0]}
                          animationDuration={600}
                        />
                      )}
                    </BarChart>
                  )}

                  {chartType === "line" && (
                    <LineChart
                      data={aggregatedData}
                      margin={{ top: 20, right: 30, left: 20, bottom: 60 }}
                    >
                      {showGrid && (
                        <CartesianGrid
                          strokeDasharray="3 3"
                          stroke={isDark ? "#27272a" : "#e2e8f0"}
                          vertical={false}
                        />
                      )}
                      <XAxis
                        dataKey="displayName"
                        stroke={isDark ? "#a1a1aa" : "#64748b"}
                        fontSize={11}
                        interval={0}
                        angle={-25}
                        textAnchor="end"
                        tickMargin={10}
                      />
                      <YAxis
                        stroke={isDark ? "#a1a1aa" : "#64748b"}
                        fontSize={11}
                        tickFormatter={(v) =>
                          v >= 1000 ? `${(v / 1000).toFixed(0)}k` : v
                        }
                      />
                      <Tooltip
                        contentStyle={{
                          backgroundColor: isDark ? "#27272a" : "#fff",
                          color: isDark ? "#fff" : "#000",
                          borderRadius: "8px",
                          border: isDark ? "1px solid #27272a" : "1px solid #e2e8f0",
                          fontSize: "12px",
                        }}
                      />
                      {showLegend && <Legend verticalAlign="top" height={36} />}
                      <Line
                        type="monotone"
                        dataKey="value"
                        name={metricCol || "Tendencia"}
                        stroke={activePalette.primary}
                        strokeWidth={3}
                        dot={{ r: 4, fill: activePalette.primary }}
                        activeDot={{ r: 7 }}
                        animationDuration={700}
                      />
                      {secondaryMetricCol && (
                        <Line
                          type="monotone"
                          dataKey="value2"
                          name={secondaryMetricCol}
                          stroke={activePalette.secondary}
                          strokeWidth={2.5}
                          strokeDasharray="4 4"
                          dot={{ r: 3, fill: activePalette.secondary }}
                          animationDuration={700}
                        />
                      )}
                    </LineChart>
                  )}

                  {chartType === "area" && (
                    <AreaChart
                      data={aggregatedData}
                      margin={{ top: 20, right: 30, left: 20, bottom: 60 }}
                    >
                      <defs>
                        <linearGradient
                          id="areaColorGrad"
                          x1="0"
                          y1="0"
                          x2="0"
                          y2="1"
                        >
                          <stop
                            offset="5%"
                            stopColor={activePalette.primary}
                            stopOpacity={0.4}
                          />
                          <stop
                            offset="95%"
                            stopColor={activePalette.primary}
                            stopOpacity={0.0}
                          />
                        </linearGradient>
                      </defs>
                      {showGrid && (
                        <CartesianGrid
                          strokeDasharray="3 3"
                          stroke={isDark ? "#27272a" : "#e2e8f0"}
                          vertical={false}
                        />
                      )}
                      <XAxis
                        dataKey="displayName"
                        stroke={isDark ? "#a1a1aa" : "#64748b"}
                        fontSize={11}
                        interval={0}
                        angle={-25}
                        textAnchor="end"
                        tickMargin={10}
                      />
                      <YAxis
                        stroke={isDark ? "#a1a1aa" : "#64748b"}
                        fontSize={11}
                        tickFormatter={(v) =>
                          v >= 1000 ? `${(v / 1000).toFixed(0)}k` : v
                        }
                      />
                      <Tooltip
                        contentStyle={{
                          backgroundColor: isDark ? "#27272a" : "#fff",
                          color: isDark ? "#fff" : "#000",
                          borderRadius: "8px",
                          border: isDark ? "1px solid #27272a" : "1px solid #e2e8f0",
                          fontSize: "12px",
                        }}
                      />
                      {showLegend && <Legend verticalAlign="top" height={36} />}
                      <Area
                        type="monotone"
                        dataKey="value"
                        name={metricCol || "Área"}
                        stroke={activePalette.primary}
                        strokeWidth={2.5}
                        fillOpacity={1}
                        fill="url(#areaColorGrad)"
                        animationDuration={700}
                      />
                    </AreaChart>
                  )}

                  {(chartType === "pie" || chartType === "donut") && (
                    <PieChart>
                      <Tooltip
                        contentStyle={{
                          backgroundColor: isDark ? "#27272a" : "#fff",
                          color: isDark ? "#fff" : "#000",
                          borderRadius: "8px",
                          border: isDark ? "1px solid #27272a" : "1px solid #e2e8f0",
                          fontSize: "12px",
                        }}
                      />
                      {showLegend && (
                        <Legend
                          layout="horizontal"
                          verticalAlign="bottom"
                          align="center"
                        />
                      )}
                      <Pie
                        data={aggregatedData}
                        dataKey="value"
                        nameKey="displayName"
                        cx="50%"
                        cy="50%"
                        innerRadius={chartType === "donut" ? 70 : 0}
                        outerRadius={120}
                        paddingAngle={chartType === "donut" ? 3 : 1}
                        animationDuration={700}
                      >
                        {aggregatedData.map((entry, index) => (
                          <Cell
                            key={`cell-${index}`}
                            fill={
                              activePalette.colors[
                              index % activePalette.colors.length
                              ]
                            }
                            stroke={isDark ? "#18181b" : "#ffffff"}
                            strokeWidth={2}
                          />
                        ))}
                      </Pie>
                    </PieChart>
                  )}

                  {chartType === "composed" && (
                    <ComposedChart
                      data={aggregatedData}
                      margin={{ top: 20, right: 30, left: 20, bottom: 60 }}
                    >
                      {showGrid && (
                        <CartesianGrid
                          strokeDasharray="3 3"
                          stroke={isDark ? "#27272a" : "#e2e8f0"}
                          vertical={false}
                        />
                      )}
                      <XAxis
                        dataKey="displayName"
                        stroke={isDark ? "#a1a1aa" : "#64748b"}
                        fontSize={11}
                        interval={0}
                        angle={-25}
                        textAnchor="end"
                        tickMargin={10}
                      />
                      <YAxis
                        stroke={isDark ? "#a1a1aa" : "#64748b"}
                        fontSize={11}
                        tickFormatter={(v) =>
                          v >= 1000 ? `${(v / 1000).toFixed(0)}k` : v
                        }
                      />
                      <Tooltip
                        contentStyle={{
                          backgroundColor: isDark ? "#27272a" : "#fff",
                          color: isDark ? "#fff" : "#000",
                          borderRadius: "8px",
                          border: isDark ? "1px solid #27272a" : "1px solid #e2e8f0",
                          fontSize: "12px",
                        }}
                      />
                      {showLegend && <Legend verticalAlign="top" height={36} />}
                      <Bar
                        dataKey="value"
                        name={metricCol || "Barras"}
                        fill={activePalette.primary}
                        radius={[6, 6, 0, 0]}
                      />
                      <Line
                        type="monotone"
                        dataKey={secondaryMetricCol ? "value2" : "value"}
                        name={
                          secondaryMetricCol
                            ? secondaryMetricCol
                            : "Tendencia Promedio"
                        }
                        stroke={
                          secondaryMetricCol
                            ? activePalette.secondary
                            : "#f59e0b"
                        }
                        strokeWidth={3}
                        dot={{ r: 4 }}
                      />
                    </ComposedChart>
                  )}
                </ResponsiveContainer>
              </div>
            ) : (
              <div className="h-[300px] flex flex-col items-center justify-center text-slate-400 dark:text-zinc-500 text-xs">
                <BarChart3 className="w-10 h-10 mb-2 opacity-30" />
                <p className="font-medium text-slate-500 dark:text-zinc-300">
                  No se pudieron generar datos para esta combinación
                </p>
                <p className="text-[11px] text-slate-400 dark:text-zinc-500 mt-1">
                  Verifique que la columna seleccionada tenga registros válidos o
                  seleccione "Conteo de filas".
                </p>
              </div>
            )}

          </div>
        </div>
      )}
    </div>
  );
}
